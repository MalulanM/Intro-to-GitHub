

Use LibraryDB
go
-- Create Authors table
CREATE TABLE Authors (
    AuthorID INT PRIMARY KEY,              -- Primary Key
    FirstName VARCHAR(50) NOT NULL,         -- Cannot be NULL
    LastName VARCHAR(50) NOT NULL,
    Email VARCHAR(100) UNIQUE,              -- Must be unique
    BirthYear INT CHECK (BirthYear >= 1900) -- Constraint: year must be >= 1900
);

-- Create Books table
CREATE TABLE Books (
    BookID INT PRIMARY KEY,                 -- Primary Key
    Title VARCHAR(100) NOT NULL,
    Genre VARCHAR(50) DEFAULT 'Unknown',    -- Default value if not provided
    Price DECIMAL(8,2) CHECK (Price >= 0),  -- Price must be non-negative
    AuthorID INT NOT NULL,                  -- Foreign Key reference
    CONSTRAINT fk_author
        FOREIGN KEY (AuthorID) REFERENCES Authors(AuthorID)
        ON UPDATE CASCADE                   -- Update AuthorID in Books if changed in Authors
       -- ON DELETE RESTRICt                 -- Prevent deletion if books exist
);

/*
Explanation of Constraints
PRIMARY KEY → Ensures each row is uniquely identified.
NOT NULL → Field must have a value.
UNIQUE → No duplicate values allowed.
CHECK → Enforces a condition on the column.
DEFAULT → Assigns a default value if none is provided.
FOREIGN KEY → Links one table to another, ensuring referential integrity.
ON UPDATE CASCADE → Updates foreign key values automatically if the parent changes.
ON DELETE RESTRICT → Prevents deletion of a parent row if child rows exist.
*/
